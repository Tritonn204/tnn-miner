cmake_minimum_required(VERSION 3.20)
project(hiveos_reporting_fixture LANGUAGES CXX)
find_package(Threads REQUIRED)
set(BOOST_SOURCE_ROOT "" CACHE PATH "Optional modular Boost source tree")
if(BOOST_SOURCE_ROOT)
    file(GLOB Boost_INCLUDE_DIRS "${BOOST_SOURCE_ROOT}/libs/*/include")
else()
    find_package(Boost 1.75 REQUIRED)
endif()
add_executable(stats_fixture fixture/stats.cpp ../../src/broadcast/broadcastServer.cpp)
target_compile_features(stats_fixture PRIVATE cxx_std_20)
target_compile_definitions(stats_fixture PRIVATE TNN_HIP BOOST_ALL_NO_LIB BOOST_ERROR_CODE_HEADER_ONLY)
target_include_directories(stats_fixture PRIVATE fixture ../../include ../../src ${Boost_INCLUDE_DIRS})
target_link_libraries(stats_fixture PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(stats_fixture PRIVATE ws2_32)
endif()
